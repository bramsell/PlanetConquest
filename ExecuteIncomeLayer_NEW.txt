void AAITeamController::ExecuteIncomeLayer()
{
	// Layer 2: Income Generation - Hierarchical resource acquisition system  
	// TODO: Implement subpriorities 2.1, 2.2, 2.3a, 2.3b
	
	// For now, keep the old Territory layer logic as a stub
	// Economic actions: Trade for black substrate if low
	if (BlackSubstrate < 200)
	{
		UE_LOG(LogTemp, Warning, TEXT("AI Team %d: P2 Income - Low black substrate (%d < 200), attempting to trade for 1000 black"),
			(int32)ControlledTeam, BlackSubstrate);
		
		bool bTradeSuccess = RequestTradeFromAI(false, 1000);
		if (bTradeSuccess)
		{
			UE_LOG(LogTemp, Warning, TEXT("AI Team %d: P2 Income - Trade successful! Now have %d black"),
				(int32)ControlledTeam, BlackSubstrate);
		}
	}
	
	// Get territory radius from GameMode
	float BaseTerritoryRadius = 5000.0f;
	if (APlanetConquestGameMode* GameMode = Cast<APlanetConquestGameMode>(GetWorld()->GetAuthGameMode()))
	{
		BaseTerritoryRadius = GameMode->TERRITORY_RADIUS;
	}
	
	float ExpansionRadius = BaseTerritoryRadius * TerritoryExpansionMultiplier;
	
	// Get all controlled cities
	TArray<ACityActor*> Cities = GetControlledCities();
if (Cities.Num() == 0) return;
	
	// Get all resources in world
	TArray<AActor*> AllResourceActors;
	UGameplayStatics::GetAllActorsOfClass(GetWorld(), AResourceActor::StaticClass(), AllResourceActors);
	
	// Find NEUTRAL resources within expansion radius
	TArray<AResourceActor*> UnclaimedResourcesInTerritory;
	for (AActor* Actor : AllResourceActors)
	{
		AResourceActor* Resource = Cast<AResourceActor>(Actor);
		if (!Resource || Resource->OwnerTeam != EOwnerTeam::Neutral) continue;
		
		// Find closest city and check if within expansion radius
		float ClosestCityDistance = FLT_MAX;
		for (ACityActor* City : Cities)
		{
			float Distance = FVector::Dist(City->GetActorLocation(), Resource->GetActorLocation());
			if (Distance < ClosestCityDistance)
			{
				ClosestCityDistance = Distance;
			}
		}
		
		if (ClosestCityDistance <= ExpansionRadius)
		{
			UnclaimedResourcesInTerritory.Add(Resource);
		}
	}
	
	if (UnclaimedResourcesInTerritory.Num() > 0)
	{
		UE_LOG(LogTemp, Log, TEXT("AI Team %d: P2 Income - Found %d unclaimed resources within %.0f radius"),
			(int32)ControlledTeam, UnclaimedResourcesInTerritory.Num(), ExpansionRadius);
		
		// Process closest resources (max 3 vehicles)
		const int32 MaxVehicles = 3;
		if (Priority2Vehicles.Num() < MaxVehicles)
		{
			ACityActor* ClosestCity = Cities[0];
			for (AResourceActor* Resource : UnclaimedResourcesInTerritory)
			{
				if (Priority2Vehicles.Num() >= MaxVehicles) break;
				
				TArray<AResourceActor*> SingleResource;
				SingleResource.Add(Resource);
				EnsureResourceCapture(SingleResource, ClosestCity, 2);
			}
		}
	}
}
