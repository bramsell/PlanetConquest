The core problem is that your avoidance system is fighting itself. You have three separate avoidance systems (coastline following, obstacle sweeps, and lateral checks) all running simultaneously and producing conflicting steering directions. The coastline follower says "go left," the obstacle avoidance says "go right," and the vehicle oscillates or orbits.

Here's what's specifically going wrong:

**Orbiting around targets near water:** The coastline follower enters `bFollowingCoastline = true` and steers perpendicular to the coast. But the target is on the other side of the water, so the vehicle follows the coastline indefinitely, circling around. The exit condition (`!bWaterAhead`) only checks directly forward — if the vehicle is moving parallel to the coast, forward is always clear, but it never turns *toward* the target.

**Vehicles moving the wrong direction:** When a vehicle detects water ahead, it immediately enters coastline mode and picks a perpendicular direction. But it checks water at only one point ahead (`LookAheadTime` distance). If there's a small inlet or narrow water feature, it triggers coastline mode unnecessarily and sends the vehicle sideways.

**Mine mesh trapping:** The mine spawns with collision, and your sweep radius is 2.5x the collision sphere. The vehicle's own mine (or a nearby one) shows up as an obstacle in the sweep, and the avoidance steers away from it, trapping the vehicle in a loop. You filter out `AResourceActor` but not `AMineActor`.

**Performance with raycasts:** Every vehicle runs multiple `SweepSingleByChannel` calls every frame — forward, left 45, left 90, right 45, right 90, plus the steering angle loop tests. With 100 vehicles that's 500+ sweeps per frame.

I'd suggest replacing all of this with a simpler potential field approach. No raycasts, no state machines, no coastline following mode. Just sample the terrain ahead and steer toward the clearest path:

```cpp
FVector AVehicleActor::CalculateSteeringDirection(FVector CurrentDirection, FVector TargetDirection, float DeltaTime)
{
    if (!GetWorld() || !OwningPlanet) return TargetDirection;
    
    FVector ForwardOnSurface = GetActorForwardVector();
    ForwardOnSurface = (ForwardOnSurface - CurrentDirection * FVector::DotProduct(ForwardOnSurface, CurrentDirection)).GetSafeNormal();
    
    // Sample terrain in a fan of directions ahead of the vehicle
    // No raycasts — just check IsPointOnLand at sample points
    
    float BestScore = -FLT_MAX;
    FVector BestDirection = TargetDirection;
    
    const int32 NumSamples = 7; // -90, -60, -30, 0, +30, +60, +90
    const float AngleStep = 30.0f;
    const float CheckDistance = FMath::Max(MovementSpeed * 1.5f, 3000.0f); // Look ahead ~1.5 seconds
    
    FVector RotationAxis = CurrentDirection; // Planet surface normal
    
    for (int32 i = 0; i < NumSamples; i++)
    {
        float AngleDegrees = -90.0f + (i * AngleStep);
        float AngleRadians = FMath::DegreesToRadians(AngleDegrees);
        
        // Rotate forward direction around surface normal
        FQuat Rotation(RotationAxis, AngleRadians);
        FVector TestForward = Rotation.RotateVector(ForwardOnSurface);
        
        // Sample 3 points along this direction (near, mid, far)
        // to detect both immediate and upcoming obstacles
        float Distances[] = { CheckDistance * 0.3f, CheckDistance * 0.6f, CheckDistance * 1.0f };
        float TerrainScore = 0.0f;
        bool bBlocked = false;
        
        for (int32 d = 0; d < 3; d++)
        {
            float SampleDist = Distances[d];
            float AngularDist = SampleDist / PlanetRadius;
            FVector SampleRotAxis = FVector::CrossProduct(CurrentDirection, TestForward).GetSafeNormal();
            
            if (SampleRotAxis.IsNearlyZero()) continue;
            
            FQuat SampleRot(SampleRotAxis, AngularDist);
            FVector SampleDirection = SampleRot.RotateVector(CurrentDirection);
            
            if (OwningPlanet->IsPointOnLand(SampleDirection))
            {
                TerrainScore += 1.0f; // Land is good
            }
            else
            {
                // Water penalty increases with proximity
                // Near water = heavily penalized, far water = mild penalty
                float ProximityWeight = (d == 0) ? 3.0f : (d == 1) ? 2.0f : 1.0f;
                TerrainScore -= ProximityWeight;
                
                if (d == 0) bBlocked = true; // Immediate water = blocked
            }
        }
        
        // Check for nearby cities/obstacles WITHOUT raycasts
        // Just check distance to known obstacles
        float ObstacleScore = 0.0f;
        float AngularDist = (CheckDistance * 0.5f) / PlanetRadius;
        FVector SampleRotAxis = FVector::CrossProduct(CurrentDirection, TestForward).GetSafeNormal();
        if (!SampleRotAxis.IsNearlyZero())
        {
            FQuat SampleRot(SampleRotAxis, AngularDist);
            FVector SampleWorldPos = PlanetCenter + SampleRot.RotateVector(CurrentDirection) * PlanetRadius;
            
            // Check distance to nearby cities
            for (ACityActor* City : OwningPlanet->SpawnedCities)
            {
                if (!City) continue;
                float DistToCity = FVector::Dist(SampleWorldPos, City->GetActorLocation());
                if (DistToCity < CityAvoidanceRadius) // e.g. 800 units
                {
                    // Penalty proportional to closeness
                    ObstacleScore -= (CityAvoidanceRadius - DistToCity) / CityAvoidanceRadius * 5.0f;
                }
            }
            
            // Check volcanoes
            for (const FVector& VolcanoPos : OwningPlanet->VolcanoPositions)
            {
                FVector VolcanoWorldPos = PlanetCenter + VolcanoPos.GetSafeNormal() * PlanetRadius;
                float DistToVolcano = FVector::Dist(SampleWorldPos, VolcanoWorldPos);
                if (DistToVolcano < 5000.0f)
                {
                    ObstacleScore -= (5000.0f - DistToVolcano) / 5000.0f * 5.0f;
                }
            }
        }
        
        // Alignment with target direction (want to go toward destination)
        float Alignment = FVector::DotProduct(TestForward, TargetDirection);
        float AlignmentScore = Alignment * 3.0f; // Weight alignment moderately
        
        // Combine scores
        float TotalScore = TerrainScore + ObstacleScore + AlignmentScore;
        
        // Penalize extreme angles (prefer gentle turns)
        float AnglePenalty = FMath::Abs(AngleDegrees) / 90.0f * 0.5f;
        TotalScore -= AnglePenalty;
        
        if (TotalScore > BestScore)
        {
            BestScore = TotalScore;
            BestDirection = TestForward;
        }
    }
    
    // Smooth steering to prevent jitter
    if (!PreviousSteeringDirection.IsNearlyZero())
    {
        float SmoothingFactor = FMath::Clamp(DeltaTime * 5.0f, 0.1f, 0.4f);
        BestDirection = FMath::Lerp(PreviousSteeringDirection, BestDirection, SmoothingFactor).GetSafeNormal();
    }
    PreviousSteeringDirection = BestDirection;
    
    return BestDirection;
}
```

This replaces all three avoidance systems with a single unified one. The key differences:

No raycasts at all — it checks `IsPointOnLand` (a noise function evaluation, very cheap) and distance to known obstacle positions (cities, volcanoes). With 100 vehicles, this is 700 noise samples + distance checks per frame instead of 500+ physics sweeps.

No state machine — there's no `bFollowingCoastline` mode that can get stuck. Every frame, the vehicle evaluates all directions fresh and picks the best one. If the best path around water is to go left, it goes left. If next frame the geometry changes (different position), it re-evaluates. No mode to get trapped in.

No orbiting — the alignment score with `TargetDirection` ensures the vehicle always biases toward its destination. If water is blocking the direct path, it picks the direction that's both clear AND closest to the target. It naturally follows coastlines when needed (the terrain score guides it along land) but always seeks to close distance to the target.

For the mine trapping issue, add `AMineActor` to your ignore list in any remaining sweep code, or better yet, set mines to a collision channel that vehicles don't query against.

You'll also want to remove the `bFollowingCoastline`, `bIsBackingUp`, and `bIsMovingPerpendicular` state variables and all the code that references them — they're no longer needed.

**Navigation mesh as a precomputed graph.** At world startup (after terrain generation), sample the planet surface on a grid and build a graph of walkable nodes. You already have `IsPointOnLand` — just sample it on an even grid across the sphere:

```cpp
// In PlanetActor, after terrain generation
void APlanetActor::GenerateNavGraph()
{
    NavNodes.Empty();
    NavEdges.Empty();
    
    // Sample points using a subdivided icosphere for even distribution
    // ~2000-5000 nodes gives good coverage without being expensive
    const int32 NumLatitudes = 60;  // 3-degree spacing
    const int32 NumLongitudes = 120; // 3-degree spacing at equator
    
    // Generate nodes
    for (int32 Lat = 0; Lat <= NumLatitudes; Lat++)
    {
        float Phi = PI * Lat / NumLatitudes; // 0 to PI
        float Z = FMath::Cos(Phi);
        float RingRadius = FMath::Sin(Phi);
        
        // Fewer samples near poles where ring radius is small
        int32 RingNodes = FMath::Max(4, FMath::RoundToInt(NumLongitudes * RingRadius));
        
        for (int32 Lon = 0; Lon < RingNodes; Lon++)
        {
            float Theta = 2.0f * PI * Lon / RingNodes;
            
            FVector Direction(
                RingRadius * FMath::Cos(Theta),
                RingRadius * FMath::Sin(Theta),
                Z
            );
            Direction.Normalize();
            
            // Only add nodes on land
            if (!IsPointOnLand(Direction)) continue;
            
            FNavNode Node;
            Node.Position = Direction; // Unit sphere position
            Node.WorldPosition = GetActorLocation() + Direction * PlanetRadius;
            Node.LandmassID = -1; // Assigned during flood fill
            Node.Index = NavNodes.Num();
            NavNodes.Add(Node);
        }
    }
    
    // Build edges: connect each node to nearby nodes
    // Use spatial hashing or just angular distance threshold
    const float MaxEdgeAngle = FMath::DegreesToRadians(5.0f); // Connect nodes within 5 degrees
    
    for (int32 i = 0; i < NavNodes.Num(); i++)
    {
        for (int32 j = i + 1; j < NavNodes.Num(); j++)
        {
            float Angle = FMath::Acos(FMath::Clamp(
                FVector::DotProduct(NavNodes[i].Position, NavNodes[j].Position), -1.0f, 1.0f));
            
            if (Angle > MaxEdgeAngle) continue;
            
            // Verify the path between nodes doesn't cross water
            // Sample a few points along the great circle arc
            bool bPathClear = true;
            const int32 NumSamples = 3;
            for (int32 s = 1; s < NumSamples; s++)
            {
                float T = (float)s / NumSamples;
                FVector MidPoint = FMath::Lerp(NavNodes[i].Position, NavNodes[j].Position, T).GetSafeNormal();
                if (!IsPointOnLand(MidPoint))
                {
                    bPathClear = false;
                    break;
                }
            }
            
            if (bPathClear)
            {
                float Distance = Angle * PlanetRadius; // Arc length
                NavNodes[i].Neighbors.Add(FNavEdge(j, Distance));
                NavNodes[j].Neighbors.Add(FNavEdge(i, Distance));
            }
        }
    }
    
    // Flood fill to assign landmass IDs (for connectivity check)
    int32 CurrentLandmassID = 0;
    for (int32 i = 0; i < NavNodes.Num(); i++)
    {
        if (NavNodes[i].LandmassID >= 0) continue; // Already assigned
        
        // BFS flood fill
        TQueue<int32> Queue;
        Queue.Enqueue(i);
        NavNodes[i].LandmassID = CurrentLandmassID;
        
        while (!Queue.IsEmpty())
        {
            int32 Current;
            Queue.Dequeue(Current);
            
            for (const FNavEdge& Edge : NavNodes[Current].Neighbors)
            {
                if (NavNodes[Edge.TargetIndex].LandmassID < 0)
                {
                    NavNodes[Edge.TargetIndex].LandmassID = CurrentLandmassID;
                    Queue.Enqueue(Edge.TargetIndex);
                }
            }
        }
        
        CurrentLandmassID++;
    }
    
    LandmassCount = CurrentLandmassID;
    UE_LOG(LogTemp, Log, TEXT("NavGraph: %d nodes, %d landmasses"), NavNodes.Num(), LandmassCount);
}
```

**Connectivity check is now instant:** When the player clicks a destination, find the nearest nav node to the vehicle and nearest nav node to the target. If they have the same `LandmassID`, the target is reachable. If not, reject the command immediately.

**Pathfinding with A*:** Standard A* on the nav graph, using great-circle distance as the heuristic:

```cpp
bool APlanetActor::FindPath(FVector StartWorldPos, FVector EndWorldPos, TArray<FVector>& OutPath)
{
    // Find nearest nav nodes
    int32 StartNode = FindNearestNavNode(StartWorldPos);
    int32 EndNode = FindNearestNavNode(EndWorldPos);
    
    if (StartNode < 0 || EndNode < 0) return false;
    
    // Quick connectivity check
    if (NavNodes[StartNode].LandmassID != NavNodes[EndNode].LandmassID)
    {
        return false; // Unreachable
    }
    
    // A* pathfinding
    TArray<float> GScore;
    TArray<float> FScore;
    TArray<int32> CameFrom;
    GScore.SetNumZeroed(NavNodes.Num());
    FScore.SetNumZeroed(NavNodes.Num());
    CameFrom.SetNum(NavNodes.Num());
    
    for (int32 i = 0; i < NavNodes.Num(); i++)
    {
        GScore[i] = FLT_MAX;
        FScore[i] = FLT_MAX;
        CameFrom[i] = -1;
    }
    
    GScore[StartNode] = 0;
    FScore[StartNode] = HeuristicDistance(StartNode, EndNode);
    
    // Priority queue (simple array-based for now)
    TSet<int32> OpenSet;
    TSet<int32> ClosedSet;
    OpenSet.Add(StartNode);
    
    while (OpenSet.Num() > 0)
    {
        // Find node with lowest FScore in open set
        int32 Current = -1;
        float LowestF = FLT_MAX;
        for (int32 Node : OpenSet)
        {
            if (FScore[Node] < LowestF)
            {
                LowestF = FScore[Node];
                Current = Node;
            }
        }
        
        if (Current == EndNode)
        {
            // Reconstruct path
            OutPath.Empty();
            int32 PathNode = EndNode;
            while (PathNode >= 0)
            {
                OutPath.Insert(NavNodes[PathNode].WorldPosition, 0);
                PathNode = CameFrom[PathNode];
            }
            // Add actual start and end positions (not just nearest nodes)
            OutPath[0] = StartWorldPos;
            OutPath.Add(EndWorldPos);
            return true;
        }
        
        OpenSet.Remove(Current);
        ClosedSet.Add(Current);
        
        for (const FNavEdge& Edge : NavNodes[Current].Neighbors)
        {
            if (ClosedSet.Contains(Edge.TargetIndex)) continue;
            
            float TentativeG = GScore[Current] + Edge.Distance;
            
            if (TentativeG < GScore[Edge.TargetIndex])
            {
                CameFrom[Edge.TargetIndex] = Current;
                GScore[Edge.TargetIndex] = TentativeG;
                FScore[Edge.TargetIndex] = TentativeG + HeuristicDistance(Edge.TargetIndex, EndNode);
                OpenSet.Add(Edge.TargetIndex);
            }
        }
    }
    
    return false; // No path found
}

float APlanetActor::HeuristicDistance(int32 NodeA, int32 NodeB)
{
    float Dot = FVector::DotProduct(NavNodes[NodeA].Position, NavNodes[NodeB].Position);
    return FMath::Acos(FMath::Clamp(Dot, -1.0f, 1.0f)) * PlanetRadius;
}

int32 APlanetActor::FindNearestNavNode(FVector WorldPos)
{
    FVector Direction = (WorldPos - GetActorLocation()).GetSafeNormal();
    
    int32 BestNode = -1;
    float BestDot = -2.0f;
    
    for (int32 i = 0; i < NavNodes.Num(); i++)
    {
        float Dot = FVector::DotProduct(Direction, NavNodes[i].Position);
        if (Dot > BestDot)
        {
            BestDot = Dot;
            BestNode = i;
        }
    }
    
    return BestNode;
}
```

{

    {

            return GetActorLocation();
